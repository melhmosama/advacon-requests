-- ============================================================================ -- pkg7_warehouse_mgmt.sql — Warehouse management RPCs (3 NEW) -- 1. warehouses_list() READ (admin + storekeeper) -- 2. admin_set_warehouse_open() WRITE (admin) + audit -- 3. admin_set_warehouse_active() WRITE (admin) + strict guards + audit -- -- Design (consultant-locked): -- Active = warehouse exists & usable in the system. -- Open = operationally ready for movements. -- Re-activating (false→true) does NOT auto-open (stays is_open=false). -- Closing an open warehouse is always allowed (stops movements, not empties). -- Opening an INACTIVE warehouse is rejected. -- Deactivating requires: closed + on_hand=0 + reserved=0 (item_warehouse rows -- with zero balance are fine — they are historical links, not stock). -- Never DELETE a warehouse. Create-only, SECURITY DEFINER, fixed search_path. -- ============================================================================ -- ───────────────────────────────────────────────────────────── -- 1. warehouses_list() — management list (independent of dashboard) -- ───────────────────────────────────────────────────────────── DROP FUNCTION IF EXISTS public.warehouses_list(); CREATE OR REPLACE FUNCTION public.warehouses_list() RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$ DECLARE v_role text; v_rows jsonb; BEGIN v_role := public.get_my_role(); IF v_role IS NULL OR v_role NOT IN ('admin','storekeeper') THEN RAISE EXCEPTION 'not_authorized'; END IF; SELECT COALESCE(jsonb_agg(to_jsonb(x) ORDER BY x.sort_order, x.code), '[]'::jsonb) INTO v_rows FROM ( SELECT w.id AS warehouse_id, w.code AS code, w.name_ar AS name_ar, w.name_en AS name_en, w.is_active AS is_active, w.is_open AS is_open, w.sort_order AS sort_order, COALESCE(sh.shelves_count, 0) AS shelves_count, COALESCE(iw.items_count, 0) AS items_count, COALESCE(iw.on_hand, 0) AS on_hand, COALESCE(iw.reserved, 0) AS reserved FROM public.warehouses w LEFT JOIN LATERAL ( SELECT COUNT(*) AS shelves_count FROM public.shelves s WHERE s.warehouse_id = w.id AND s.is_active = true ) sh ON true LEFT JOIN LATERAL ( SELECT COUNT(*) AS items_count, SUM(iw2.on_hand) AS on_hand, SUM(iw2.reserved) AS reserved FROM public.item_warehouse iw2 WHERE iw2.warehouse_id = w.id AND iw2.is_active = true ) iw ON true ) x; RETURN v_rows; END; $$; REVOKE EXECUTE ON FUNCTION public.warehouses_list() FROM PUBLIC; REVOKE EXECUTE ON FUNCTION public.warehouses_list() FROM anon; GRANT EXECUTE ON FUNCTION public.warehouses_list() TO authenticated; -- ───────────────────────────────────────────────────────────── -- 2. admin_set_warehouse_open(p_warehouse_id, p_is_open) -- Open requires the warehouse to be active. Close is always allowed. -- ───────────────────────────────────────────────────────────── DROP FUNCTION IF EXISTS public.admin_set_warehouse_open(uuid, boolean); CREATE OR REPLACE FUNCTION public.admin_set_warehouse_open( p_warehouse_id uuid, p_is_open boolean ) RETURNS uuid LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, auth, pg_temp AS $$ DECLARE v_actor uuid := auth.uid(); v_active boolean; v_open boolean; BEGIN PERFORM public.require_admin(); IF p_is_open IS NULL THEN RAISE EXCEPTION 'is_open_required'; END IF; SELECT is_active, is_open INTO v_active, v_open FROM public.warehouses WHERE id = p_warehouse_id FOR UPDATE; IF v_active IS NULL THEN RAISE EXCEPTION 'warehouse_not_found'; END IF; -- opening an inactive warehouse is rejected; closing is always allowed IF p_is_open = true AND v_active = false THEN RAISE EXCEPTION 'warehouse_inactive'; END IF; -- no-op guard (idempotent, no audit noise) IF v_open = p_is_open THEN RETURN p_warehouse_id; END IF; UPDATE public.warehouses SET is_open = p_is_open, updated_at = now(), updated_by = v_actor WHERE id = p_warehouse_id; INSERT INTO public.audit_log (actor_user_id, action, target_table, target_id, warehouse_id, summary, details) VALUES (v_actor, CASE WHEN p_is_open THEN 'warehouse.open' ELSE 'warehouse.close' END, 'warehouses', p_warehouse_id, p_warehouse_id, CASE WHEN p_is_open THEN 'Opened warehouse' ELSE 'Closed warehouse' END, jsonb_build_object('is_open', jsonb_build_object('old', v_open, 'new', p_is_open))); RETURN p_warehouse_id; END; $$; REVOKE EXECUTE ON FUNCTION public.admin_set_warehouse_open(uuid, boolean) FROM PUBLIC; REVOKE EXECUTE ON FUNCTION public.admin_set_warehouse_open(uuid, boolean) FROM anon; GRANT EXECUTE ON FUNCTION public.admin_set_warehouse_open(uuid, boolean) TO authenticated; -- ───────────────────────────────────────────────────────────── -- 3. admin_set_warehouse_active(p_warehouse_id, p_is_active) -- Activate: direct (does NOT auto-open — stays closed). -- Deactivate: requires closed + on_hand=0 + reserved=0. -- ───────────────────────────────────────────────────────────── DROP FUNCTION IF EXISTS public.admin_set_warehouse_active(uuid, boolean); CREATE OR REPLACE FUNCTION public.admin_set_warehouse_active( p_warehouse_id uuid, p_is_active boolean ) RETURNS uuid LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, auth, pg_temp AS $$ DECLARE v_actor uuid := auth.uid(); v_active boolean; v_open boolean; v_on_hand numeric; v_reserved numeric; BEGIN PERFORM public.require_admin(); IF p_is_active IS NULL THEN RAISE EXCEPTION 'is_active_required'; END IF; -- 1. warehouse exists (lock) SELECT is_active, is_open INTO v_active, v_open FROM public.warehouses WHERE id = p_warehouse_id FOR UPDATE; IF v_active IS NULL THEN RAISE EXCEPTION 'warehouse_not_found'; END IF; -- no-op guard IF v_active = p_is_active THEN RETURN p_warehouse_id; END IF; -- 2. if deactivating → strict guards IF p_is_active = false THEN IF v_open = true THEN RAISE EXCEPTION 'warehouse_must_be_closed_first'; END IF; SELECT COALESCE(SUM(on_hand), 0), COALESCE(SUM(reserved), 0) INTO v_on_hand, v_reserved FROM public.item_warehouse WHERE warehouse_id = p_warehouse_id; IF v_reserved <> 0 THEN RAISE EXCEPTION 'warehouse_has_reservations'; END IF; IF v_on_hand <> 0 THEN RAISE EXCEPTION 'warehouse_has_stock'; END IF; END IF; -- activating (false→true): is_active=true and is_open is forced to false -- 3. update -- 3. update — activating always forces is_open=false (Active != Open invariant) UPDATE public.warehouses SET is_active = p_is_active, is_open = CASE WHEN p_is_active = true THEN false ELSE is_open END, updated_at = now(), updated_by = v_actor WHERE id = p_warehouse_id; -- 4. audit old→new (warehouse_id for easy filtering) INSERT INTO public.audit_log (actor_user_id, action, target_table, target_id, warehouse_id, summary, details) VALUES (v_actor, CASE WHEN p_is_active THEN 'warehouse.activate' ELSE 'warehouse.deactivate' END, 'warehouses', p_warehouse_id, p_warehouse_id, CASE WHEN p_is_active THEN 'Activated warehouse' ELSE 'Deactivated warehouse' END, jsonb_build_object('is_active', jsonb_build_object('old', v_active, 'new', p_is_active))); RETURN p_warehouse_id; END; $$; REVOKE EXECUTE ON FUNCTION public.admin_set_warehouse_active(uuid, boolean) FROM PUBLIC; REVOKE EXECUTE ON FUNCTION public.admin_set_warehouse_active(uuid, boolean) FROM anon; GRANT EXECUTE ON FUNCTION public.admin_set_warehouse_active(uuid, boolean) TO authenticated;
